# funciones.py:
def prueba(a,b): # -> define la función prueba que recibe los parámetros a y b
    a = a + 1 # -> incrementa el valor de a, le suma 1
    b = b + 2 # -> incrementa el valor de b, le suma 2
    return(a,b) # -> retorna los valores de a y b
